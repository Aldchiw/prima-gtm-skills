// Ingests the "value_chain_map" source (output/value_chain_map.csv, filtered
// to icp_status=="IN") into Supabase "accounts" + "contacts", deduping
// against accounts that already exist there. Same safe pattern as
// scripts/sync-supabase.js: parseCSV/loadEnv reused verbatim, dry-run by
// default, assertNoProtectedColumns() enforced in code.
//
// ============================== REGLA DE ORO ==============================
// Two totally different write shapes depending on whether the account
// already exists in Supabase:
//
// 1. MERGE (account_name matches an existing account): update ONLY
//    accounts.source / market_segment / tier / icp_status on that row.
//    Every other column on that account -- domain, company_category,
//    sub_segment, priority, vertical_owner, excluded, exclusion_reason,
//    scope_tier, needs_manual_scope_confirmation, signal_*, assigned_user_id,
//    id, created_at -- is PROTECTED and never touched. Nothing in `contacts`
//    for that account is touched either -- not inserted, not updated. An
//    existing account's contact data is none of this script's business.
//
// 2. INSERT (account_name has no match): a brand-new accounts row with
//    account_name, source='value_chain_map', market_segment, tier,
//    icp_status, company_category, and domain (only if the batch/sample
//    contact data happened to carry one). assigned_user_id is left NULL on
//    every insert -- see "On assigned_user_id" below, this script never sets
//    it. If this account also has a named contact in
//    value_chain_batch.csv/value_chain_sample.csv, INSERT that contact too
//    (contact_name, contact_title, linkedin_url, contact_email,
//    email_status, stage='Not Contacted' -- the one place `stage` is ever
//    written, same exception sync-supabase.js documents: a brand-new contact
//    row has no existing cockpit-owned value to clobber). An IN account with
//    no contact row at all is correct and expected -- it surfaces as "Needs
//    contact" in the cockpit, exactly like the ~38 TAM placeholders from the
//    orphan-contact fix.
//
// assertNoProtectedColumns() enforces both shapes in code: the merge payload
// is checked against PROTECTED_ACCOUNT_COLUMNS_ON_MERGE (every real accounts
// column except the 4 new ones), and the contact-insert payload is checked
// against PROTECTED_CONTACT_COLUMNS (everything cockpit-owned, `stage`
// excepted for the insert-only reason above).
//
// On assigned_user_id: DELIBERATELY never set by this script, insert or
// merge. `owner_vertical` from value_chain_map.csv is read and reported
// (so a human can sanity-check Aldahir/Manu counts) but never written
// anywhere -- the cockpit assigns real ownership via its own deterministic
// per-category rule, not this ingestion pass.
// ============================================================================
//
// DEDUP -- exact match first, then a validated safe-prefix heuristic, never a
// loose fuzzy match. This project already hit a false positive from loose
// token-overlap matching once (building value_chain_map.csv originally
// matched "Mitsubishi Power" to "Mitsubishi Electric Power Products" --
// two different entities in the same conglomerate -- because a naive
// normalizer stripped "Power" as if it were a generic suffix word). The fix
// kept here: normalize() strips ONLY true legal-entity suffixes (inc, corp,
// company, llc, ltd, group) -- never sector words like Power/Energy/Systems/
// Technologies/Solutions, which are load-bearing in this domain's company
// names. A prefix match is only trusted (auto-MERGE) when the shorter side's
// normalized form is 4+ characters, exactly like the value_chain_map.csv
// build script. Below that length (in practice, only "Eos" in this dataset),
// a candidate prefix hit is too collision-prone to trust blindly -- it's
// reported as AMBIGUOUS instead, never auto-merged or auto-inserted, so a
// human decides (see spec's explicit Eos/Mainspring callout). Mainspring
// itself is 10 characters and clears the normal prefix-match bar on its own.
//
// Usage:
//   node scripts/ingest-value-chain.js            (DRY RUN -- default, no writes)
//   node scripts/ingest-value-chain.js --apply    (performs the real writes)
//
// Requires: .env in repo root with SUPABASE_URL and SUPABASE_SERVICE_ROLE,
// and @supabase/supabase-js installed.

'use strict';
const fs = require('fs');
const path = require('path');

const MAP_CSV_PATH = path.join(__dirname, '..', 'output', 'value_chain_map.csv');
const BATCH_CSV_PATH = path.join(__dirname, '..', 'output', 'value_chain_batch.csv');
const SAMPLE_CSV_PATH = path.join(__dirname, '..', 'output', 'value_chain_sample.csv');
const ENV_PATH = path.join(__dirname, '..', '.env');
const APPLY = process.argv.includes('--apply');
const SELECT_CHUNK_SIZE = 200; // keep .in() filter lists well under URL length limits

// The 4 new columns this script is allowed to write on an EXISTING (merged)
// account. Nothing else, ever, on a merge.
const ACCOUNT_MERGE_COLUMNS = ['source', 'market_segment', 'tier', 'icp_status'];

// Every other real column on accounts -- protected on merge, no exceptions.
// (id/assigned_user_id/created_at are always protected, insert or merge --
// see PROTECTED_ACCOUNT_COLUMNS_ALWAYS below.)
const PROTECTED_ACCOUNT_COLUMNS_ON_MERGE = [
  'id', 'account_name', 'domain', 'company_category', 'sub_segment', 'priority',
  'vertical_owner', 'excluded', 'exclusion_reason', 'scope_tier',
  'needs_manual_scope_confirmation', 'signal_source', 'signal_detail',
  'signal_url', 'signal_date', 'assigned_user_id', 'created_at',
];

// Allowed on a brand-new contact INSERT only. `stage` is the one field this
// script is allowed to set -- but only here, only on insert, per the same
// exception scripts/sync-supabase.js documents.
const CONTACT_INSERT_COLUMNS = ['contact_name', 'contact_title', 'linkedin_url', 'contact_email', 'email_status', 'stage'];
// Never in a contact payload, under any circumstance (this script never
// updates an existing contact at all, but this stays as a hard check on the
// insert payload too, since a bug that let last_touch_at/team_status/notes
// slip into a brand-new row would still be a REGLA DE ORO violation).
const PROTECTED_CONTACT_COLUMNS_ALWAYS = ['id', 'account_id', 'last_touch_at', 'team_status', 'notes', 'created_at'];

const VALID_COMPANY_CATEGORIES = ['Cat1', 'Cat2', 'Cat3', 'Cat4'];

// --------------------------- CSV parsing (reused verbatim) ---------------------------
function parseCSV(text) {
  const rows = [];
  let row = [];
  let field = '';
  let inQuotes = false;
  let i = 0;
  const n = text.length;
  while (i < n) {
    const c = text[i];
    if (inQuotes) {
      if (c === '"') {
        if (text[i + 1] === '"') { field += '"'; i += 2; continue; }
        inQuotes = false; i++; continue;
      }
      field += c; i++; continue;
    }
    if (c === '"') { inQuotes = true; i++; continue; }
    if (c === ',') { row.push(field); field = ''; i++; continue; }
    if (c === '\r') { i++; continue; }
    if (c === '\n') {
      row.push(field); field = '';
      rows.push(row); row = [];
      i++; continue;
    }
    field += c; i++;
  }
  if (field.length > 0 || row.length > 0) { row.push(field); rows.push(row); }
  return rows.filter(r => !(r.length === 1 && r[0] === ''));
}

// ------------------------------ .env parsing (reused verbatim) ---------------------------------
function loadEnv() {
  if (!fs.existsSync(ENV_PATH)) {
    throw new Error('.env no encontrado en la raíz del repo -- crea SUPABASE_URL y SUPABASE_SERVICE_ROLE ahí primero.');
  }
  const raw = fs.readFileSync(ENV_PATH, 'utf8').replace(/^﻿/, '');
  const env = {};
  for (const line of raw.split(/\r?\n/)) {
    const trimmed = line.trim();
    if (!trimmed || trimmed.startsWith('#')) continue;
    const eq = trimmed.indexOf('=');
    if (eq === -1) continue;
    const key = trimmed.slice(0, eq).trim();
    let value = trimmed.slice(eq + 1).trim();
    if ((value.startsWith('"') && value.endsWith('"')) || (value.startsWith("'") && value.endsWith("'"))) {
      value = value.slice(1, -1);
    }
    env[key] = value;
  }
  if (!env.SUPABASE_URL) throw new Error('.env sin SUPABASE_URL.');
  if (!env.SUPABASE_SERVICE_ROLE) throw new Error('.env sin SUPABASE_SERVICE_ROLE.');
  // Never log env.SUPABASE_SERVICE_ROLE -- not even in errors below.
  return env;
}

function toRecords(csvRows) {
  const header = csvRows[0];
  const records = [];
  for (let i = 1; i < csvRows.length; i++) {
    const row = csvRows[i];
    const rec = {};
    header.forEach((col, idx) => { rec[col] = (row[idx] !== undefined ? row[idx] : '').trim(); });
    records.push(rec);
  }
  return { header, records };
}

function readCSVFile(filePath) {
  const raw = fs.readFileSync(filePath, 'utf8').replace(/^﻿/, '');
  const csvRows = parseCSV(raw);
  if (csvRows.length === 0) return { header: [], records: [] };
  return toRecords(csvRows);
}

// Hard runtime guard for the REGLA DE ORO -- throws if a payload about to be
// sent anywhere near a write ever contains a protected key.
function assertNoProtectedColumns(payload, protectedCols, context) {
  const hit = Object.keys(payload).filter(k => protectedCols.includes(k));
  if (hit.length > 0) {
    throw new Error(`REGLA DE ORO violada en ${context}: columnas protegidas en payload: ${hit.join(', ')}`);
  }
}

function chunk(arr, size) {
  const out = [];
  for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size));
  return out;
}

// Normalize for matching: lowercase, strip only true legal-entity suffixes.
// Deliberately does NOT strip sector words (Power/Energy/Systems/
// Technologies/Solutions) -- see the false-positive lesson in the header
// comment above.
function normalize(name) {
  return String(name || '')
    .toLowerCase()
    .replace(/\([^)]*\)/g, '')
    .replace(/[.,]/g, '')
    .replace(/\b(inc|corp|corporation|company|co|llc|ltd|group)\b/g, '')
    .replace(/\s+/g, ' ')
    .trim();
}

// Returns { kind: 'exact' } | { kind: 'prefix', match } | { kind: 'ambiguous_short', match } | { kind: 'none' }
function matchAgainstExisting(mapName, existingIndex) {
  const rawLower = mapName.toLowerCase().trim();
  for (const e of existingIndex) {
    if (e.raw.toLowerCase() === rawLower) return { kind: 'exact', match: e.raw, id: e.id };
  }
  const n = normalize(mapName);
  if (!n) return { kind: 'none' };
  for (const e of existingIndex) {
    if (e.norm === n) return { kind: 'exact', match: e.raw, id: e.id };
  }
  // Prefix-at-word-boundary, either direction. Only trusted (auto-MERGE)
  // when BOTH sides are 4+ chars -- a short side (e.g. "eos", 3 chars) is
  // too collision-prone to trust blindly, so it's downgraded to
  // "ambiguous_short" instead of an auto-merge.
  for (const e of existingIndex) {
    if (e.norm.length < 4 || n.length < 4) continue;
    if (e.norm.startsWith(n + ' ') || n.startsWith(e.norm + ' ')) {
      return { kind: 'prefix', match: e.raw, id: e.id };
    }
  }
  // Short-name second pass: same prefix check, but with the length gate
  // lifted, ONLY to detect a plausible-but-unconfirmed candidate -- never
  // auto-acted on, always reported as ambiguous per the spec's explicit
  // Eos/Mainspring callout.
  if (n.length > 0 && n.length < 4) {
    for (const e of existingIndex) {
      if (e.norm.startsWith(n + ' ')) return { kind: 'ambiguous_short', match: e.raw, id: e.id };
    }
  }
  return { kind: 'none' };
}

async function main() {
  const env = loadEnv();
  const { createClient } = require('@supabase/supabase-js');
  const supabase = createClient(env.SUPABASE_URL, env.SUPABASE_SERVICE_ROLE, {
    auth: { persistSession: false },
  });

  // ---------------------------- Load the 3 source CSVs ----------------------------
  const mapData = readCSVFile(MAP_CSV_PATH);
  if (!mapData.records.length) throw new Error('value_chain_map.csv vacío -- nada que ingerir.');
  const requiredMapCols = ['account_name', 'source', 'market_segment', 'tier', 'company_category', 'sub_segment_draft', 'icp_status', 'owner_vertical', 'dup_flag', 'reason'];
  const missingMapCols = requiredMapCols.filter(c => !mapData.header.includes(c));
  if (missingMapCols.length) throw new Error(`value_chain_map.csv le faltan columnas: ${missingMapCols.join(', ')}`);

  const contactRecords = [
    ...(fs.existsSync(BATCH_CSV_PATH) ? readCSVFile(BATCH_CSV_PATH).records : []),
    ...(fs.existsSync(SAMPLE_CSV_PATH) ? readCSVFile(SAMPLE_CSV_PATH).records : []),
  ];
  const contactByAccountName = new Map();
  for (const rec of contactRecords) {
    if (!rec.account_name) continue;
    if (!rec.contact_name || !rec.contact_name.trim()) continue; // no real name -- not a contact row
    // If the same account_name appears in both files (shouldn't, batch was
    // built excluding sample accounts), last one wins -- consistent with the
    // "last row wins" convention sync-supabase.js already uses.
    contactByAccountName.set(rec.account_name, rec);
  }

  // ---------------------------- Filter map to icp_status IN ----------------------------
  const skippedNoAccountName = [];
  const skippedBadCategory = [];
  const inRows = [];
  mapData.records.forEach((rec, idx) => {
    const rowNum = idx + 2;
    if (rec.icp_status !== 'IN') return; // MARGINAL/OUT are out of scope for this ingestion
    if (!rec.account_name) {
      skippedNoAccountName.push({ row: rowNum });
      return;
    }
    if (!VALID_COMPANY_CATEGORIES.includes(rec.company_category)) {
      skippedBadCategory.push({ row: rowNum, account_name: rec.account_name, company_category: rec.company_category });
      return;
    }
    inRows.push(rec);
  });

  // ------------------------- Fetch existing accounts (live, not the CSV's own dup_flag) -------------------------
  // Deliberately re-checks against live Supabase rather than trusting
  // value_chain_map.csv's own dup_flag column -- that column was computed
  // against local CSV snapshots (leads_master.csv / needs_contact.csv) at
  // build time, which can lag behind what's actually in Supabase (this is
  // exactly how "Mainspring" was missed originally: not an algorithm gap,
  // the account just wasn't present in the local snapshot checked then).
  const { data: allAccounts, error: accErr } = await supabase.from('accounts').select('id, account_name');
  if (accErr) throw new Error(`Error leyendo accounts existentes: ${accErr.message}`);
  const existingIndex = (allAccounts || []).map(a => ({ raw: a.account_name, norm: normalize(a.account_name), id: a.id }));

  // ---------------------------- Classify each IN row ----------------------------
  const toMerge = [];    // [{ account_name, matched_existing_name, id, payload }]
  const toInsertAccount = []; // [{ account_name, payload, contact }]
  const ambiguous = [];  // [{ account_name, candidate_match }]

  for (const rec of inRows) {
    const contact = contactByAccountName.get(rec.account_name) || null;
    const m = matchAgainstExisting(rec.account_name, existingIndex);

    if (m.kind === 'ambiguous_short') {
      ambiguous.push({ account_name: rec.account_name, candidate_match: m.match, reason: 'nombre corto (<4 caracteres normalizado) -- match no confiable, requiere confirmación humana' });
      continue;
    }

    if (m.kind === 'exact' || m.kind === 'prefix') {
      const payload = {
        source: 'value_chain_map',
        market_segment: rec.market_segment,
        tier: rec.tier,
        icp_status: rec.icp_status,
        updated_at: new Date().toISOString(),
      };
      assertNoProtectedColumns(payload, PROTECTED_ACCOUNT_COLUMNS_ON_MERGE, `accounts MERGE payload for "${rec.account_name}"`);
      toMerge.push({ account_name: rec.account_name, matched_existing_name: m.match, id: m.id, payload, match_kind: m.kind, owner_vertical_fyi: rec.owner_vertical });
      continue;
    }

    // No match at all -> brand-new account.
    const accountPayload = {
      account_name: rec.account_name,
      source: 'value_chain_map',
      market_segment: rec.market_segment,
      tier: rec.tier,
      icp_status: rec.icp_status,
      company_category: rec.company_category,
      domain: contact ? (contact.domain || null) : null,
    };
    toInsertAccount.push({ account_name: rec.account_name, payload: accountPayload, contact, owner_vertical_fyi: rec.owner_vertical });
  }

  // ----------------------------- Apply (if --apply) -----------------------------
  const newAccountIdByName = new Map();
  if (APPLY) {
    for (const { id, payload, account_name } of toMerge) {
      const { error } = await supabase.from('accounts').update(payload).eq('id', id);
      if (error) throw new Error(`Error en MERGE de account "${account_name}": ${error.message}`);
    }
    if (toInsertAccount.length > 0) {
      const rowsToInsert = toInsertAccount.map(x => x.payload);
      for (const batch of chunk(rowsToInsert, SELECT_CHUNK_SIZE)) {
        const { data, error } = await supabase.from('accounts').insert(batch).select('id, account_name');
        if (error) throw new Error(`Error insertando accounts nuevas: ${error.message}`);
        for (const row of data || []) newAccountIdByName.set(row.account_name, row.id);
      }
    }
    for (const { account_name, contact } of toInsertAccount) {
      if (!contact) continue;
      const accountId = newAccountIdByName.get(account_name);
      if (!accountId) throw new Error(`No se resolvió account_id recién insertado para "${account_name}" -- no se puede insertar su contacto.`);
      const contactPayload = {
        contact_name: contact.contact_name,
        contact_title: contact.contact_title || '',
        linkedin_url: contact.linkedin_url || '',
        contact_email: contact.contact_email || null,
        email_status: contact.email_status || 'NOT_FOUND',
        stage: 'Not Contacted', // the ONE field this script ever sets -- brand-new row only
      };
      assertNoProtectedColumns(contactPayload, PROTECTED_CONTACT_COLUMNS_ALWAYS, `contacts INSERT payload for "${account_name}"`);
      const { error } = await supabase.from('contacts').insert({ account_id: accountId, ...contactPayload });
      if (error) throw new Error(`Error insertando contact para "${account_name}": ${error.message}`);
    }
  } else {
    // Dry-run: still assert the contact payload shape for every account that
    // would get one, so the report below is a true preview of what --apply
    // would do (and so assertNoProtectedColumns has actually run over every
    // payload, not just the account ones).
    for (const { account_name, contact } of toInsertAccount) {
      if (!contact) continue;
      const contactPayload = {
        contact_name: contact.contact_name,
        contact_title: contact.contact_title || '',
        linkedin_url: contact.linkedin_url || '',
        contact_email: contact.contact_email || null,
        email_status: contact.email_status || 'NOT_FOUND',
        stage: 'Not Contacted',
      };
      assertNoProtectedColumns(contactPayload, PROTECTED_CONTACT_COLUMNS_ALWAYS, `contacts INSERT payload (dry-run) for "${account_name}"`);
    }
  }

  // --------------------------------- Report ---------------------------------
  const insertsWithContact = toInsertAccount.filter(x => x.contact);
  const insertsNoContact = toInsertAccount.filter(x => !x.contact);

  console.log(APPLY ? '=== MODO APPLY (writes reales ejecutados) ===' : '=== MODO DRY-RUN (nada escrito) ===');
  console.log('value_chain_map.csv filas totales:', mapData.records.length, '| icp_status==IN:', inRows.length);
  console.log('');
  console.log('Accounts a INSERTAR (nuevas):', toInsertAccount.length);
  console.log('  - con contacto (INSERT en contacts):', insertsWithContact.length);
  console.log('  - sin contacto (quedan "Needs contact" en el cockpit):', insertsNoContact.length);
  console.log('Accounts a MERGEAR (dedup, ya existen):', toMerge.length);
  console.log('Ambiguas (nombre corto, NO se toca, requiere revisión humana):', ambiguous.length);
  console.log('');
  console.log('Filas saltadas por account_name vacío:', skippedNoAccountName.length);
  if (skippedNoAccountName.length) console.log(JSON.stringify(skippedNoAccountName));
  console.log('Filas saltadas por company_category inválida (no Cat1-4):', skippedBadCategory.length);
  if (skippedBadCategory.length) console.log(JSON.stringify(skippedBadCategory));
  console.log('');
  console.log('=== Detalle MERGE (dedup) -- verificar contra la lista conocida de 14 ===');
  toMerge.forEach(m => console.log(`  [${m.match_kind}] "${m.account_name}" -> cuenta existente "${m.matched_existing_name}" (owner_vertical del map, solo informativo: ${m.owner_vertical_fyi})`));
  console.log('');
  console.log('=== Detalle AMBIGUAS (reportadas, NO insertadas NI mergeadas) ===');
  ambiguous.forEach(a => console.log(`  "${a.account_name}" -> posible match "${a.candidate_match}" (${a.reason})`));
  console.log('');
  console.log('Columnas protegidas en MERGE -- NUNCA tocadas (confirmado por assertNoProtectedColumns):');
  console.log('  ', PROTECTED_ACCOUNT_COLUMNS_ON_MERGE.join(', '));
  console.log('Columnas protegidas en contacts -- NUNCA tocadas salvo `stage` en INSERT de fila nueva:');
  console.log('  ', PROTECTED_CONTACT_COLUMNS_ALWAYS.join(', '));
  console.log('assigned_user_id: nunca escrito por este script (ni insert ni merge) -- lo asigna el cockpit.');
  if (!APPLY) {
    console.log('');
    console.log('Este fue un DRY RUN. Nada se escribió en Supabase. Corre con --apply para ejecutar los writes de verdad.');
  }
}

if (require.main === module) {
  main().then(() => process.exit(0)).catch(e => {
    // Never include env values in error output.
    console.error('ERROR:', e.message);
    process.exit(1);
  });
}

module.exports = {
  parseCSV, loadEnv, normalize, matchAgainstExisting,
  ACCOUNT_MERGE_COLUMNS, PROTECTED_ACCOUNT_COLUMNS_ON_MERGE,
  CONTACT_INSERT_COLUMNS, PROTECTED_CONTACT_COLUMNS_ALWAYS,
};
